## Stage-Wise Reasoning Trace Decomposition

**Priority:** high
**Module:** intent
**Risk:** medium
**Paper:** ClinHallu: A Benchmark for Diagnosing Stage-Wise Hallucinations in Medical MLLM Reasoning

### Description
Decomposes complex reasoning into structured stages (Visual Recognition, Knowledge Recall, Reasoning Integration) with explicit traces for each stage. Uses stage-replacement interventions to measure how correcting specific stages affects final outcomes.

### How to Apply
Could significantly improve agos debugging and reliability by decomposing agent reasoning into traceable stages, making it easier to identify where errors occur in complex multi-step tasks. This would enhance the system's ability to self-diagnose and recover from reasoning failures.

### Implementation Hint
Extend the Intent Engine to generate structured reasoning traces with explicit stages (perception, knowledge retrieval, integration). Add intervention capabilities to the Agent Kernel for stage-level debugging and correction during execution.

#### Stage-wise Reasoning Validation Pipeline
Breaks down complex reasoning into discrete stages (visual recognition, knowledge recall, reasoning) and validates each stage separately. This allows for fine-grained error detection and correction in multi-step agent reasoning.
```python
class ReasoningValidator:
    def __init__(self, judge_api):
        self.judge_api = judge_api
        self.stage_map = {
            "perception": ["visual_recognition"],
            "knowledge": ["knowledge_recall"], 
            "reasoning": ["reasoning"]
        }
    
    async def validate_reasoning_chain(self, task_context, reasoning_steps, reference_steps=None):
        """Validate each stage of reasoning against reference or ground truth"""
        results = {}
        for stage_name, step_keys in self.stage_map.items():
            for step_key in step_keys:
                candidate_text = self._extract_step_text(reasoning_steps, step_key)
                reference_text = self._extract_step_text(reference_steps, step_key) if reference_steps else None
                
                validation = await self._judge_step_hallucination(
                    task_context, candidate_text, reference_text, step_key
                )
                results[f"{stage_name}_{step_key}"] = validation
        return results
    
    async def _judge_step_hallucination(self, context, candidate, reference, step_type):
        """Judge if a reasoning step contains hallucinations"""
        prompt = self._build_validation_prompt(context, candidate, reference, step_type)
        response = await self.judge_api.generate_async(prompt)
        return self._parse_validation_response(response)
```

#### Hierarchical Task State Tracking
Maintains detailed state for each task including generation status, validation results, and completion tracking. Enables robust resume/retry logic and progress monitoring for long-running agent tasks.
```python
class TaskStateManager:
    def __init__(self, storage_path):
        self.storage_path = Path(storage_path)
        self.completed_cache = {}
    
    async def load_completed_task_ids(self, task_type: str) -> set:
        """Load set of completed task IDs to avoid reprocessing"""
        cache_key = f"{task_type}_completed"
        if cache_key in self.completed_cache:
            return self.completed_cache[cache_key]
        
        completed_file = self.storage_path / f"{task_type}_completed.json"
        if completed_file.exists():
            with open(completed_file) as f:
                completed_ids = set(json.load(f))
        else:
            completed_ids = set()
        
        self.completed_cache[cache_key] = completed_ids
        return completed_ids
    
    async def mark_task_completed(self, task_type: str, task_id: str, result: dict):
        """Mark task as completed and persist state"""
        completed_ids = await self.load_completed_task_ids(task_type)
        completed_ids.add(task_id)
        
        # Update cache
        self.completed_cache[f"{task_type}_completed"] = completed_ids
        
        # Persist to disk
        completed_file = self.storage_path / f"{task_type}_completed.json"
        with open(completed_file, 'w') as f:
            json.dump(list(completed_ids), f)
        
        # Store detailed result
        result_file = self.storage_path / f"{task_type}_results.jsonl"
        with open(result_file, 'a') as f:
            json.dump({"task_id": task_id, "result": result, "timestamp": time.time()}, f)
            f.write('\n')
```

#### Multi-Target Generation Pipeline
Supports multiple generation targets (gold, model, replace experiments) with different prompts and contexts. Enables A/B testing of agent strategies and gradual capability replacement for iterative improvement.
```python
class MultiTargetGenerator:
    def __init__(self, api_wrapper):
        self.api_wrapper = api_wrapper
        self.target_configs = {
            "baseline": {"mode": "standard", "prefix_steps": []},
            "replace_perception": {"mode": "continue_from_perception", "prefix_steps": [("perception", "")]},
            "replace_knowledge": {"mode": "continue_from_knowledge", "prefix_steps": [("perception", ""), ("knowledge", "")]},
            "replace_reasoning": {"mode": "continue_from_reasoning", "prefix_steps": [("perception", ""), ("knowledge", ""), ("reasoning", "")]}
        }
    
    async def generate_for_target(self, task_context, target: str, reference_steps=None):
        """Generate response for specific target configuration"""
        config = self.target_configs.get(target, self.target_configs["baseline"])
        
        # Build prefix from reference if doing replacement
        prefix_steps = []
        if reference_steps and config["prefix_steps"]:
            for step_name, _ in config["prefix_steps"]:
                if step_name in reference_steps:
                    prefix_steps.append((step_name, reference_steps[step_name]))
        
        prompt = self._build_generation_prompt(task_context, config["mode"], prefix_steps)
        raw_response = await self.api_wrapper.generate_async(prompt)
        
        # Assemble full response with prefix
        full_response = self._assemble_response(prefix_steps, raw_response)
        return self._parse_structured_response(full_response)
    
    def _assemble_response(self, prefix_steps, raw_response):
        """Combine prefix steps with new generation"""
        prefix_blocks = [f"[{name}]\n{text}" for name, text in prefix_steps if text.strip()]
        if not prefix_blocks:
            return raw_response
        return "\n\n".join(prefix_blocks) + "\n\n" + raw_response
```

#### Concurrent Task Execution with Backoff
Executes multiple tasks concurrently with intelligent retry logic, rate limiting, and graceful degradation. Essential for scalable agent coordination and robust task processing.
```python
class ConcurrentTaskExecutor:
    def __init__(self, max_workers=8, retry_attempts=3):
        self.max_workers = max_workers
        self.retry_attempts = retry_attempts
    
    async def execute_batch(self, tasks, processor_func, progress_callback=None):
        """Execute batch of tasks with concurrency control and retries"""
        results = []
        failed_tasks = []
        
        with ThreadPoolExecutor(max_workers=self.max_workers) as executor:
            # Submit all tasks
            future_to_task = {
                executor.submit(self._execute_with_retry, processor_func, task): task 
                for task in tasks
            }
            
            completed = 0
            for future in as_completed(future_to_task):
                task = future_to_task[future]
                try:
                    result = future.result()
                    results.append(result)
                    if progress_callback:
                        await progress_callback(completed, len(tasks), result)
                except Exception as error:
                    logger.error(f"Task failed after retries: {error}")
                    failed_tasks.append((task, str(error)))
                
                completed += 1
        
        return results, failed_tasks
    
    def _execute_with_retry(self, processor_func, task):
        """Execute single task with exponential backoff retry"""
        last_error = None
        for attempt in range(self.retry_attempts + 1):
            try:
                return processor_func(task)
            except Exception as error:
                last_error = error
                if attempt < self.retry_attempts:
                    delay = (2 ** attempt) + random.uniform(0, 1)
                    time.sleep(delay)
                    logger.warning(f"Retry {attempt + 1}/{self.retry_attempts} for task after {delay:.1f}s")
        raise last_error
```

#### Structured Response Parsing with Fallbacks
Robust parsing of LLM responses with fallback strategies for malformed JSON, code blocks, and missing fields. Critical for reliable agent-to-agent communication and tool integration.
```python
class StructuredResponseParser:
    def __init__(self, required_fields=None, fallback_values=None):
        self.required_fields = required_fields or []
        self.fallback_values = fallback_values or {}
    
    async def parse_llm_response(self, raw_response: str, response_type="json"):
        """Parse LLM response with multiple fallback strategies"""
        cleaned = self._clean_response(raw_response)
        
        if response_type == "json":
            return self._parse_json_response(cleaned)
        elif response_type == "structured":
            return self._parse_structured_response(cleaned)
        else:
            return {"content": cleaned}
    
    def _clean_response(self, raw_response: str) -> str:
        """Remove code blocks and clean formatting"""
        cleaned = raw_response.strip()
        
        # Remove markdown code blocks
        if cleaned.startswith("```"):
            lines = cleaned.split("\n")
            lines = [line for line in lines if not line.strip().startswith("```")]
            cleaned = "\n".join(lines).strip()
        
        return cleaned
    
    def _parse_json_response(self, cleaned: str) -> dict:
        """Parse JSON with fallback handling"""
        try:
            result = json.loads(cleaned)
        except json.JSONDecodeError as error:
            logger.warning(f"JSON parse failed: {error}")
            # Fallback: try to extract JSON from text
            result = self._extract_json_from_text(cleaned)
        
        # Ensure required fields exist
        for field in self.required_fields:
            if field not in result:
                result[field] = self.fallback_values.get(field, None)
        
        return result
    
    def _extract_json_from_text(self, text: str) -> dict:
        """Extract JSON object from mixed text content"""
        # Look for JSON-like patterns
        json_pattern = r'\{[^{}]*\}'
        matches = re.findall(json_pattern, text)
        
        for match in matches:
            try:
                return json.loads(match)
            except json.JSONDecodeError:
                continue
        
        # Final fallback
        return {"raw_content": text, "parse_error": True}
```

#### Config-Driven Model Registry
Flexible model configuration system supporting multiple APIs, model variants, and purpose-specific routing. Enables dynamic model selection and easy experimentation with different LLM backends.
```python
class ModelRegistry:
    def __init__(self, config_path):
        self.config = self._load_config(config_path)
        self._api_cache = {}
    
    async def resolve_model_for_purpose(self, model_ref: str, purpose: str = "default"):
        """Resolve model configuration for specific purpose (judge, generation, etc)"""
        registry = self._get_registry_for_purpose(purpose)
        
        if model_ref in registry:
            spec = registry[model_ref].copy()
            spec.setdefault("model_name", model_ref)
            spec.setdefault("registry_key", model_ref)
            return spec
        
        # Fallback to default registry
        if purpose != "default":
            default_registry = self._get_registry_for_purpose("default")
            if model_ref in default_registry:
                spec = default_registry[model_ref].copy()
                spec.setdefault("model_name", model_ref)
                return spec
        
        # Final fallback
        return {"model_name": model_ref, "registry_key": model_ref}
    
    async def get_api_wrapper(self, model_ref: str, purpose: str = "default"):
        """Get cached API wrapper for model"""
        cache_key = f"{model_ref}_{purpose}"
        
        if cache_key in self._api_cache:
            return self._api_cache[cache_key]
        
        config = await self.resolve_model_for_purpose(model_ref, purpose)
        wrapper = self._create_api_wrapper(config)
        self._api_cache[cache_key] = wrapper
        return wrapper
    
    def _get_registry_for_purpose(self, purpose: str) -> dict:
        """Get model registry for specific purpose"""
        if purpose == "judge":
            return self.config.get("judge_registry", {})
        return self.config.get("model_registry", {})
    
    def _create_api_wrapper(self, config: dict):
        """Create API wrapper from configuration"""
        return CoTAPIWrapper(
            base_url=config.get("base_url", "https://api.openai.com/v1"),
            api_key=config.get("api_key", ""),
            model_name=config["model_name"],
            max_tokens=config.get("max_tokens", 2048),
            temperature=config.get("temperature", 0.1)
        )
```
