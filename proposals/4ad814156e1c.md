## Training-Free Diverse Sampling Methods

**Priority:** medium
**Module:** intent
**Risk:** high
**Paper:** Wildfire Spread Scenarios: Increasing Sample Diversity of Segmentation Diffusion Models with Training-Free Methods

### Description
Techniques like particle guidance, SPELL, and clustering-based methods that increase sample diversity from diffusion models without requiring additional training. These methods help models generate multiple plausible outcomes for uncertain scenarios while being computationally efficient.

### How to Apply
The Intent Engine could use diverse sampling when generating multiple execution plans for ambiguous requests, helping agos consider various valid interpretations of user intent. The Knowledge System could benefit when retrieving semantically similar but contextually different pieces of information.

#### Multi-Modal Sampling Strategy Pattern
Dynamically switches between different sampling strategies (naive, particle guidance, SPELL) based on diversity metrics and performance feedback. Allows pluggable strategy selection for different contexts.
```python
import asyncio
from abc import ABC, abstractmethod
from typing import Dict, Any, Optional

class SamplingStrategy(ABC):
    @abstractmethod
    async def sample(self, context: Dict[str, Any]) -> Dict[str, Any]:
        pass

class AdaptiveSamplingManager:
    def __init__(self):
        self.strategies = {}
        self.performance_history = {}
        self.current_strategy = None
    
    def register_strategy(self, name: str, strategy: SamplingStrategy):
        self.strategies[name] = strategy
        self.performance_history[name] = []
    
    async def adaptive_sample(self, context: Dict[str, Any], 
                            diversity_threshold: float = 0.5) -> Dict[str, Any]:
        # Select strategy based on recent performance and diversity needs
        if self.current_strategy is None or self._should_switch_strategy(diversity_threshold):
            self.current_strategy = self._select_best_strategy(context)
        
        result = await self.strategies[self.current_strategy].sample(context)
        self._update_performance_metrics(self.current_strategy, result)
        return result
    
    def _should_switch_strategy(self, threshold: float) -> bool:
        if not self.performance_history[self.current_strategy]:
            return False
        recent_diversity = self.performance_history[self.current_strategy][-5:]
        return sum(recent_diversity) / len(recent_diversity) < threshold
```

#### Distribution Evaluation Metrics System
Comprehensive metrics system that tracks mode diversity, distribution divergence (TVD), and cluster analysis for evaluating multi-modal outputs. Enables quantitative assessment of solution variety.
```python
import torch
import numpy as np
from typing import List, Dict, Any

class DiversityMetrics:
    def __init__(self, expected_modes: int = 10):
        self.expected_modes = expected_modes
        self.mode_history = []
    
    async def compute_solution_diversity(self, solutions: List[Dict[str, Any]]) -> Dict[str, float]:
        """Compute diversity metrics for a set of solutions"""
        # Extract solution features for clustering
        features = [self._extract_features(sol) for sol in solutions]
        
        # Compute cluster distribution
        cluster_ids = self._cluster_solutions(features)
        cluster_distribution = np.bincount(cluster_ids) / len(cluster_ids)
        
        # Compute uniform distribution for comparison
        uniform_distribution = np.ones(self.expected_modes) / self.expected_modes
        
        # Total Variation Distance from uniform
        tvd_uniform = np.abs(cluster_distribution - uniform_distribution).sum()
        
        # Count distinct modes
        n_distinct_modes = len(np.unique(cluster_ids))
        
        # Expected number of samples to see all modes (coupon collector)
        expected_coupons = self._compute_expected_coupons(cluster_distribution)
        
        metrics = {
            'tvd_uniform': tvd_uniform,
            'distinct_modes': n_distinct_modes,
            'expected_coupons': expected_coupons,
            'mode_coverage': n_distinct_modes / self.expected_modes
        }
        
        self.mode_history.append(metrics)
        return metrics
    
    def _compute_expected_coupons(self, distribution: np.ndarray) -> float:
        """Compute expected samples needed to see all modes"""
        return sum(1/p for p in distribution if p > 0)
```

#### Hierarchical Model Mixin Architecture
Uses mixin pattern to compose dataset-specific behavior with base model functionality. Enables flexible composition of capabilities without deep inheritance hierarchies.
```python
from abc import ABC, abstractmethod
from typing import Any, Dict

class TaskMixin(ABC):
    @abstractmethod
    def get_task_evaluator(self, config: Any):
        pass
    
    @abstractmethod
    def process_task_input(self, input_data: Dict[str, Any]) -> Dict[str, Any]:
        pass

class FileTaskMixin(TaskMixin):
    def get_task_evaluator(self, config: Any):
        return FileTaskEvaluator(config.get('file_patterns', []))
    
    def process_task_input(self, input_data: Dict[str, Any]) -> Dict[str, Any]:
        return {
            'files': input_data.get('files', []),
            'operations': input_data.get('operations', []),
            'workspace_path': input_data.get('workspace', '/tmp')
        }

class WebTaskMixin(TaskMixin):
    def get_task_evaluator(self, config: Any):
        return WebTaskEvaluator(config.get('allowed_domains', []))
    
    def process_task_input(self, input_data: Dict[str, Any]) -> Dict[str, Any]:
        return {
            'urls': input_data.get('urls', []),
            'headers': input_data.get('headers', {}),
            'timeout': input_data.get('timeout', 30)
        }

class BaseTaskAgent:
    def __init__(self, agent_id: str):
        self.agent_id = agent_id
        self.task_evaluator = None
    
    async def execute_task(self, task_data: Dict[str, Any]) -> Dict[str, Any]:
        processed_input = self.process_task_input(task_data)
        result = await self._execute_core_logic(processed_input)
        return self.task_evaluator.evaluate(result)

class FileAgent(FileTaskMixin, BaseTaskAgent):
    def __init__(self, agent_id: str, config: Any):
        super().__init__(agent_id)
        self.task_evaluator = self.get_task_evaluator(config)
```

#### Batch Processing with Aggregated Metrics
Processes multiple samples in batches and aggregates metrics across samples. Enables efficient evaluation of solution quality across multiple attempts while maintaining statistical significance.
```python
import asyncio
from typing import List, Dict, Any, Callable
from dataclasses import dataclass

@dataclass
class BatchResult:
    results: List[Dict[str, Any]]
    aggregated_metrics: Dict[str, float]
    batch_id: str

class BatchProcessor:
    def __init__(self, batch_size: int = 8, aggregation_strategy: str = 'mean'):
        self.batch_size = batch_size
        self.aggregation_strategy = aggregation_strategy
        self.results_history = []
    
    async def process_batch(self, 
                          tasks: List[Dict[str, Any]], 
                          processor_func: Callable,
                          metric_extractors: Dict[str, Callable]) -> BatchResult:
        """Process a batch of tasks and aggregate results"""
        
        # Process tasks in parallel batches
        batched_tasks = [tasks[i:i+self.batch_size] 
                        for i in range(0, len(tasks), self.batch_size)]
        
        all_results = []
        for batch in batched_tasks:
            batch_results = await asyncio.gather(
                *[processor_func(task) for task in batch]
            )
            all_results.extend(batch_results)
        
        # Aggregate metrics across all results
        aggregated_metrics = {}
        for metric_name, extractor in metric_extractors.items():
            values = [extractor(result) for result in all_results]
            aggregated_metrics[metric_name] = self._aggregate_values(values)
        
        batch_result = BatchResult(
            results=all_results,
            aggregated_metrics=aggregated_metrics,
            batch_id=f"batch_{len(self.results_history)}"
        )
        
        self.results_history.append(batch_result)
        return batch_result
    
    def _aggregate_values(self, values: List[float]) -> float:
        if self.aggregation_strategy == 'mean':
            return sum(values) / len(values)
        elif self.aggregation_strategy == 'max':
            return max(values)
        else:
            return sum(values) / len(values)
```

#### Training-Free Adaptation Pattern
Implements training-free methods to modify behavior through sampling strategies rather than model retraining. Enables rapid adaptation to new requirements without expensive retraining cycles.
```python
from typing import Dict, Any, Optional
import asyncio

class TrainingFreeAdapter:
    def __init__(self, base_processor):
        self.base_processor = base_processor
        self.adaptation_strategies = {}
        self.current_adaptations = {}
    
    def register_adaptation(self, name: str, adaptation_func):
        """Register a training-free adaptation strategy"""
        self.adaptation_strategies[name] = adaptation_func
    
    async def adapt_and_process(self, 
                               input_data: Dict[str, Any],
                               adaptation_config: Optional[Dict[str, Any]] = None) -> Dict[str, Any]:
        """Apply adaptations and process without retraining"""
        
        if adaptation_config:
            # Apply runtime adaptations
            adapted_processor = self._apply_adaptations(
                self.base_processor, adaptation_config
            )
        else:
            adapted_processor = self.base_processor
        
        # Process with adapted behavior
        result = await adapted_processor.process(input_data)
        
        # Apply post-processing adaptations if needed
        if 'post_process' in self.current_adaptations:
            result = await self.current_adaptations['post_process'](result)
        
        return result
    
    def _apply_adaptations(self, processor, config: Dict[str, Any]):
        """Apply adaptations to processor without modifying original"""
        adapted = processor.copy() if hasattr(processor, 'copy') else processor
        
        for adaptation_name, params in config.items():
            if adaptation_name in self.adaptation_strategies:
                adaptation_func = self.adaptation_strategies[adaptation_name]
                adapted = adaptation_func(adapted, params)
                self.current_adaptations[adaptation_name] = adaptation_func
        
        return adapted
    
    async def measure_adaptation_quality(self, 
                                       test_inputs: List[Dict[str, Any]],
                                       adaptation_config: Dict[str, Any]) -> Dict[str, float]:
        """Measure quality of adaptation without retraining"""
        results = []
        for input_data in test_inputs:
            result = await self.adapt_and_process(input_data, adaptation_config)
            results.append(result)
        
        return self._compute_adaptation_metrics(results)
```
