## Systematic Text-Space Skill Optimization

**Priority:** high
**Module:** evolution
**Risk:** high
**Paper:** SkillOpt: Executive Strategy for Self-Evolving Agent Skills

### Description
SkillOpt introduces a systematic optimizer that treats agent skills as trainable external state, using scored rollouts to generate bounded add/delete/replace edits on skill documents. The optimizer accepts edits only when they strictly improve validation scores, with textual learning-rate budgets and rejected-edit buffers for stability.

### How to Apply
This could dramatically improve agos by allowing the system to systematically evolve and optimize its internal skills (like prompt templates, tool usage patterns, coordination strategies) based on execution feedback rather than relying on static hand-crafted approaches. The technique would enable agos to continuously improve its performance across different domains through disciplined skill refinement.

### Implementation Hint
Implement a skill optimization service that tracks execution outcomes for different agent behaviors, maintains skill documents as editable text, and uses an LLM-based optimizer to propose improvements. Include validation scoring, edit acceptance criteria, and rollback mechanisms for failed optimizations.

#### Multi-Model Evaluation Pipeline
Systematic evaluation framework that trains multiple models, compares performance across metrics, and generates comprehensive reports. Could evaluate different agent strategies or tool combinations.
```python
import asyncio
from typing import Dict, Any, List
from dataclasses import dataclass

@dataclass
class EvaluationResult:
    name: str
    metrics: Dict[str, float]
    execution_time: float

class StrategyEvaluator:
    def __init__(self, output_dir: str):
        self.output_dir = output_dir
        self.strategies = {}
    
    async def evaluate_all(self, test_cases: List[Any]) -> Dict[str, EvaluationResult]:
        results = {}
        for name, strategy in self.strategies.items():
            start_time = asyncio.get_event_loop().time()
            metrics = await self._evaluate_strategy(strategy, test_cases)
            elapsed = asyncio.get_event_loop().time() - start_time
            results[name] = EvaluationResult(name, metrics, elapsed)
        return results
    
    async def _evaluate_strategy(self, strategy, test_cases) -> Dict[str, float]:
        # Evaluate strategy performance
        successes = 0
        total_time = 0
        for case in test_cases:
            success, duration = await strategy.execute(case)
            successes += int(success)
            total_time += duration
        return {
            'success_rate': successes / len(test_cases),
            'avg_time': total_time / len(test_cases)
        }
```

#### Feature Pipeline with Preprocessing
Modular data preprocessing pipeline with configurable transformations, scaling, and feature engineering. Could preprocess agent inputs and memories before storage or retrieval.
```python
import asyncio
from typing import Any, Dict, List, Optional, Callable
from dataclasses import dataclass

@dataclass
class ProcessingStep:
    name: str
    transform: Callable
    config: Dict[str, Any]

class DataPipeline:
    def __init__(self, config: Dict[str, Any]):
        self.config = config
        self.steps: List[ProcessingStep] = []
        self.feature_names: Optional[List[str]] = None
    
    async def add_step(self, name: str, transform: Callable, **kwargs):
        step = ProcessingStep(name, transform, kwargs)
        self.steps.append(step)
    
    async def process(self, data: Any) -> Any:
        """Run data through preprocessing pipeline"""
        result = data
        for step in self.steps:
            if asyncio.iscoroutinefunction(step.transform):
                result = await step.transform(result, **step.config)
            else:
                result = step.transform(result, **step.config)
        return result
    
    async def fit_transform(self, data: Any) -> Any:
        """Fit pipeline and transform data"""
        # Fit each step and transform
        result = data
        for step in self.steps:
            result = await self._fit_step(step, result)
        return result
```

#### Comprehensive Metric Collection

import asyncio
import time
from typing import Dict, Any, List, Optional
from dataclasses import dataclass, field
from collections import defaultdict
import json

@dataclass
class MetricSnapshot:
    timestamp: float
    metrics: Dict[str, float]
    context: Dict[str, Any] = field(default_factory=dict)

class MetricsCollector:
    def __init__(self, output_dir: str = 'metrics'):
        self.metrics: Dict[str, List[float]] = defaultdict(list)
        self.snapshots: List[MetricSnapshot] = []
        self.output_dir = output_dir
    
    async def record(self, metric_name: str, value: float, **context):
        """Record a metric value with optional context"""
        self.metrics[metric_name].append(value)
        snapshot = MetricSnapshot(
            timestamp=time.time(),
            metrics={metric_name: value},
            context=context
        )
        self.snapshots.append(snapshot)
    
    async def compute_stats(self, metric_name: str) -> Dict[str, float]:
        """Compute statistics for a metric"""
        values = self.metrics[metric_name]
        if not values:
            return {}
        return {
            'mean': sum(values) / len(values),
            'min': min(values),
            'max': max(values),
            'count': len(values)
        }
    
    async def export_report(self) -> Dict[str, Any]:
        """Export comprehensive metrics report"""
        report = {}
        for metric_name in self.metrics:
            report[metric_name] = await self.compute_stats(metric_name)
        return report

#### Factory Pattern for Dynamic Creation
Factory pattern for creating configured instances of different types (models, agents, tools) based on string identifiers and configuration parameters.
```python
from typing import Dict, Any, Callable, Optional
import asyncio

class ComponentFactory:
    def __init__(self):
        self._builders: Dict[str, Callable] = {}
        self._configs: Dict[str, Dict[str, Any]] = {}
    
    def register(self, component_type: str, builder: Callable, default_config: Optional[Dict] = None):
        """Register a component builder with default config"""
        self._builders[component_type] = builder
        if default_config:
            self._configs[component_type] = default_config
    
    async def create(self, component_type: str, config: Optional[Dict] = None, **kwargs) -> Any:
        """Create component instance with merged config"""
        if component_type not in self._builders:
            raise ValueError(f"Unknown component: {component_type}")
        
        # Merge configs: defaults < provided config < kwargs
        merged_config = self._configs.get(component_type, {}).copy()
        if config:
            merged_config.update(config)
        merged_config.update(kwargs)
        
        builder = self._builders[component_type]
        if asyncio.iscoroutinefunction(builder):
            return await builder(**merged_config)
        else:
            return builder(**merged_config)
    
    def available_types(self) -> List[str]:
        return list(self._builders.keys())
```

#### Timer Decorator for Performance Tracking
Decorator-based performance timing that automatically tracks function execution times. Simple yet effective for profiling system components.
```python
import asyncio
import time
from functools import wraps
from typing import Dict, List

class PerformanceTracker:
    def __init__(self):
        self.timings: Dict[str, List[float]] = {}
    
    def timer(self, func_name: Optional[str] = None):
        """Decorator to time function execution"""
        def decorator(func):
            name = func_name or f"{func.__module__}.{func.__name__}"
            
            @wraps(func)
            async def async_wrapper(*args, **kwargs):
                start = time.perf_counter()
                try:
                    result = await func(*args, **kwargs)
                    return result
                finally:
                    elapsed = time.perf_counter() - start
                    if name not in self.timings:
                        self.timings[name] = []
                    self.timings[name].append(elapsed)
            
            @wraps(func)
            def sync_wrapper(*args, **kwargs):
                start = time.perf_counter()
                try:
                    result = func(*args, **kwargs)
                    return result
                finally:
                    elapsed = time.perf_counter() - start
                    if name not in self.timings:
                        self.timings[name] = []
                    self.timings[name].append(elapsed)
            
            return async_wrapper if asyncio.iscoroutinefunction(func) else sync_wrapper
        return decorator
    
    async def get_stats(self, func_name: str) -> Dict[str, float]:
        timings = self.timings.get(func_name, [])
        if not timings:
            return {}
        return {
            'avg': sum(timings) / len(timings),
            'min': min(timings),
            'max': max(timings),
            'total': sum(timings),
            'count': len(timings)
        }
```
