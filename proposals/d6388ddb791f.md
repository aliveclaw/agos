## Persistent Visual Memory Module

**Risk:** high

### Description

#### Persistent Visual Memory Cross-Attention
Maintains a parallel cross-attention pathway to preserved visual embeddings, preventing visual information dilution during long text generation. Uses bottleneck adapters with zero-initialized gates to allow selective visual retrieval without disrupting the main autoregressive flow.
```python
import torch
import torch.nn as nn

class PersistentMemoryBlock(nn.Module):
    def __init__(self, hidden_size: int, memory_dim: int = 512):
        super().__init__()
        self.hidden_size = hidden_size
        self.memory_dim = memory_dim
        
        # Bottleneck projection
        self.down_proj = nn.Linear(hidden_size, memory_dim)
        self.memory_key_proj = nn.Linear(hidden_size, memory_dim)
        self.memory_value_proj = nn.Linear(hidden_size, memory_dim)
        
        # Cross-attention to persistent memory
        self.cross_attn = nn.MultiheadAttention(memory_dim, num_heads=8, batch_first=True)
        
        # Restoration pathway
        self.mlp = nn.Sequential(
            nn.Linear(memory_dim, memory_dim * 2),
            nn.GELU(),
            nn.Linear(memory_dim * 2, hidden_size)
        )
        
        # Zero-initialized gate
        self.gate_alpha = nn.Parameter(torch.zeros(1))
    
    async def forward(self, hidden_states: torch.Tensor, persistent_memory: torch.Tensor) -> torch.Tensor:
        # Project to lower dimension
        query = self.down_proj(hidden_states)
        memory_keys = self.memory_key_proj(persistent_memory)
        memory_values = self.memory_value_proj(persistent_memory)
        
        # Cross-attention to retrieve from persistent memory
        attended_memory, _ = self.cross_attn(query, memory_keys, memory_values)
        
        # Restore to original dimension
        memory_features = self.mlp(attended_memory)
        
        # Gated addition with zero initialization
        return hidden_states + self.gate_alpha * memory_features
```

#### Multi-Layer Memory Injection Strategy
Injects memory access at strategic layers rather than every layer, using intermediate positions to balance memory access with computational efficiency. Implements selective layer targeting based on model depth.
```python
class MemoryInjectionConfig:
    def __init__(self, total_layers: int, num_injection_points: int = 3):
        self.total_layers = total_layers
        self.injection_layers = self._calculate_injection_points(total_layers, num_injection_points)
    
    def _calculate_injection_points(self, total_layers: int, num_points: int) -> list[int]:
        """Calculate optimal layer positions for memory injection"""
        if total_layers <= 8:
            # Small models: inject at 1/3 intervals
            return [total_layers // 3, 2 * total_layers // 3, total_layers - 1]
        else:
            # Larger models: use the pattern from paper
            step = total_layers // num_points
            return [step * (i + 1) for i in range(num_points)]

class LayeredMemoryAccess:
    def __init__(self, config: MemoryInjectionConfig):
        self.injection_layers = set(config.injection_layers)
        self.persistent_memories = {}
    
    async def should_inject_memory(self, layer_idx: int) -> bool:
        return layer_idx in self.injection_layers
    
    async def store_persistent_memory(self, key: str, memory: torch.Tensor):
        self.persistent_memories[key] = memory.detach().clone()
    
    async def retrieve_memory(self, key: str) -> torch.Tensor:
        return self.persistent_memories.get(key)
```

#### Zero-Initialized Gating for Gradual Learning
Uses learnable gates initialized at zero to allow new pathways to be gradually integrated without disrupting existing functionality. Enables safe addition of new capabilities to running systems.
```python
import torch
import torch.nn as nn
from typing import Optional

class AdaptiveGate(nn.Module):
    def __init__(self, feature_dim: int, init_value: float = 0.0):
        super().__init__()
        self.gate_weight = nn.Parameter(torch.full((feature_dim,), init_value))
        self.gate_bias = nn.Parameter(torch.zeros(feature_dim))
    
    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return torch.sigmoid(self.gate_weight + self.gate_bias) * x

class GradualFeatureIntegration:
    def __init__(self, base_system, feature_dim: int):
        self.base_system = base_system
        self.gates = {}
        self.feature_dim = feature_dim
    
    async def add_feature_pathway(self, name: str, feature_processor) -> None:
        """Add new feature pathway with zero-initialized gate"""
        self.gates[name] = AdaptiveGate(self.feature_dim, init_value=0.0)
        setattr(self, f"{name}_processor", feature_processor)
    
    async def forward_with_features(self, base_input: torch.Tensor, **feature_inputs) -> torch.Tensor:
        base_output = await self.base_system(base_input)
        
        for feature_name, feature_data in feature_inputs.items():
            if feature_name in self.gates:
                processor = getattr(self, f"{feature_name}_processor")
                feature_output = await processor(feature_data)
                gated_feature = self.gates[feature_name](feature_output)
                base_output = base_output + gated_feature
        
        return base_output
```
**Integration steps:** 1. Use in evolution module to safely add new agent capabilities without breaking existing behaviors. 2. Apply to tool integration for gradual rollout of new tools with monitored adoption. 3. Implement in policy engine for gradual policy updates with safety gates.

#### Format-Constrained Reward Function
Enforces structured output formats using regex matching with lookahead assertions to prevent nested or malformed tags. Provides graduated rewards for partial compliance.
```python
import re
import asyncio
from typing import List, Dict, Any, Optional

class StructuredOutputValidator:
    def __init__(self):
        self.format_patterns = {
            'think_answer': r'^<think>(?:(?!</think>).)*?</think>\s*<answer>(?:(?!</answer>).)*?</answer>\s*$',
            'json_response': r'^\{(?:(?!\}$).)*\}$',
            'action_plan': r'^<plan>(?:(?!</plan>).)*?</plan>\s*<action>(?:(?!</action>).)*?</action>\s*$'
        }
    
    async def validate_format(self, content: str, format_type: str) -> float:
        """Returns reward score (0.0-1.0) for format compliance"""
        pattern = self.format_patterns.get(format_type)
        if not pattern:
            return 0.0
        
        if re.match(pattern, content, re.DOTALL):
            return 1.0
        
        # Partial credit for having required tags
        if format_type == 'think_answer':
            has_think = bool(re.search(r'<think>', content))
            has_answer = bool(re.search(r'<answer>', content))
            return 0.3 if has_think and has_answer else 0.1
        
        return 0.1
    
    async def extract_structured_content(self, content: str, format_type: str) -> Optional[Dict[str, str]]:
        """Extract structured parts from validated content"""
        if format_type == 'think_answer':
            think_match = re.search(r'<think>(.*?)</think>', content, re.DOTALL)
            answer_match = re.search(r'<answer>(.*?)</answer>', content, re.DOTALL)
            
            if think_match and answer_match:
                return {
                    'reasoning': think_match.group(1).strip(),
                    'answer': answer_match.group(1).strip()
                }
        
        return None

class ResponseQualityEvaluator:
    def __init__(self):
        self.validator = StructuredOutputValidator()
    
    async def evaluate_response_quality(self, responses: List[str], expected_format: str) -> List[float]:
        scores = []
        for response in responses:
            format_score = await self.validator.validate_format(response, expected_format)
            scores.append(format_score)
        return scores
```

#### Targeted Parameter Monitoring
Monitors specific model parameters (like gate values) during training/adaptation to track learning progress and identify parameter drift. Uses recursive attribute access for deep parameter inspection.
```python
import functools
import asyncio
from typing import Dict, List, Any, Optional
import logging

class ParameterMonitor:
    def __init__(self, target_params: List[str]):
        self.target_params = target_params
        self.history = {}
        self.thresholds = {}
    
    def _rgetattr(self, obj: Any, attr: str, *args) -> Any:
        """Recursively get nested attributes"""
        def _getattr(obj, attr):
            if attr.isdigit():
                return obj[int(attr)]
            return getattr(obj, attr, *args)
        return functools.reduce(_getattr, attr.split('.'), obj)
    
    async def monitor_parameters(self, model: Any, step: int) -> Dict[str, float]:
        """Monitor specified parameters and return current values"""
        current_values = {}
        
        for param_path in self.target_params:
            try:
                # Handle DDP/wrapped models
                real_model = getattr(model, 'module', model)
                
                # Clean path (remove 'model.' prefix if present)
                clean_path = param_path.replace('model.', '') if param_path.startswith('model.') else param_path
                
                param = self._rgetattr(real_model, clean_path)
                
                if hasattr(param, 'item'):
                    value = param.item()
                    current_values[param_path] = value
                    
                    # Store in history
                    if param_path not in self.history:
                        self.history[param_path] = []
                    self.history[param_path].append((step, value))
                    
            except (AttributeError, IndexError, TypeError) as e:
                logging.warning(f"Could not access parameter {param_path}: {e}")
        
        return current_values
    
    async def set_alert_threshold(self, param_path: str, min_val: float, max_val: float):
        """Set monitoring thresholds for parameter drift detection"""
        self.thresholds[param_path] = (min_val, max_val)
    
    async def check_parameter_drift(self, current_values: Dict[str, float]) -> List[str]:
        """Check for parameter drift beyond thresholds"""
        alerts = []
        for param_path, value in current_values.items():
            if param_path in self.thresholds:
                min_val, max_val = self.thresholds[param_path]
                if value < min_val or value > max_val:
                    alerts.append(f"Parameter {param_path} drift: {value} outside [{min_val}, {max_val}]")
        return alerts
```
**Integration steps:** 1. Monitor agent learning parameters during self-improvement cycles to detect convergence or divergence. 2. Track tool usage patterns and adaptation weights in the tool bus. 3. Monitor policy engine parameters to ensure gradual rather than abrupt policy changes.
